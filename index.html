Type that moves with you.

A study of scroll-driven typography with GSAP and ScrollTrigger. Each example shows the JavaScript behind it, how it becomes a Shopify section, and what a merchant can change in the theme customizer.

function splitIntro(el) {
  // Split into words, then letters. "mask" clips
  // each word so letters rise up from behind it.
  const split = SplitText.create(el,
    { type: "words,chars", mask: "words" });
  const stagger = el.dataset.stagger / 1000;

  const intro = gsap.from(split.chars, {
    yPercent: 110, opacity: 0, duration: .9,
    ease: "power4.out", stagger, delay: .3
  });

  // Scrubbed exit: words drift up as you scroll away
  gsap.to(split.words, {
    yPercent: -40, opacity: .15, ease: "none", stagger: .04,
    scrollTrigger: { trigger: el, start: "top top",
      end: "bottom top", scrub: true }
  });

  // Replay button restarts the time-based intro
  replayButton?.addEventListener("click",
    () => intro.restart(true));
}

Scrub + word stagger

Words that fill in as you read

The sentence is split into words that start faint. Scroll position drives a staggered tween, so each word darkens as you reach it. Scroll back up and they fade again.

Good motion explains the interface. It points your eye to what changed, then gets out of the way so you can keep reading.

JavaScript 
function wordFill(el) {
  const split = SplitText.create(el, { type: "words" });
  const from = el.dataset.from / 100;  // 18 → 0.18

  gsap.fromTo(split.words, { opacity: from }, {
    opacity: 1, ease: "none", stagger: .1,
    scrollTrigger: {
      trigger: el,
      start: "top 80%",   // element top hits 80% down the screen
      end: "bottom 45%",
      scrub: true         // progress = scroll position
    }
  });
}

JavaScript 
function stickyStage(stage) {
  const words  = [...stage.querySelectorAll(".word")];
  const splits = words.map(w =>
    SplitText.create(w, { type: "chars" }));

  // Hide every word after the first, flipped away
  gsap.set(splits.slice(1).map(s => s.chars),
    { rotationX: 90, opacity: 0 });

  const tl = gsap.timeline({ scrollTrigger: {
    trigger: stage, start: "top top",
    end: "bottom bottom", scrub: .5 } });

  // Works for any number of words (blocks)
  splits.forEach((s, i) => {
    if (!i) return;
    tl.to(splits[i - 1].chars, { rotationX: -90,
        opacity: 0, stagger: .03, duration: .3 }, i - .3)
      .to(s.chars, { rotationX: 0, opacity: 1,
        stagger: .03, duration: .3 }, i);
  });
}

Scrubbed xPercent

A marquee you drive with the scrollbar

Two oversized rows move in opposite directions. There is no timer or loop; each row's xPercent is tied to how far this section has scrolled through the viewport.

function marquee(row) {
  const d = +row.dataset.distance;       // e.g. 35
  const left = row.dataset.direction === "left";

  gsap.fromTo(row,
    { xPercent: left ? 0 : -d },
    { xPercent: left ? -d : 0, ease: "none",
      scrollTrigger: {
        trigger: row.closest("section"),
        start: "top bottom",  // section enters
        end: "bottom top",    // section leaves
        scrub: true
      } });
}

<scroll-anim> web component

Motion set in markup

Each card is wrapped in a small custom element that reads its motion from HTML attributes. Because it is a web component, it sets itself up and cleans itself up, with no page script to wire.

Set in markup

Motion values live in attributes like from-y and scrub, so they can come straight from block settings.

Cleans itself up

disconnectedCallback kills the tween, so the theme editor can re-render sections without leaking triggers.

Stays readable

With reduced motion on, the component does nothing and the content simply sits in place.
JavaScript 
class ScrollAnim extends HTMLElement {
  connectedCallback() {
    if (!motionOK()) return;
    const n = (a, d) => this.hasAttribute(a)
      ? parseFloat(this.getAttribute(a)) : d;

    this.tween = gsap.fromTo(this,
      { y: n("from-y", 0),
        opacity: n("from-opacity", 1) },
      { y: 0, opacity: 1, ease: "none",
        scrollTrigger: { trigger: this,
          start: this.getAttribute("start"),
          end: this.getAttribute("end"),
          scrub: n("scrub", true) } });
  }
  disconnectedCallback() {
    // Theme editor removes sections while editing
    this.tween?.scrollTrigger?.kill();
    this.tween?.kill();
  }
}
customElements.define("scroll-anim", ScrollAnim);

Theme editor wiring

Keeping it working while merchants edit

The theme editor re-renders a section every time a setting changes. Each section's animations are built inside its own gsap.matchMedia() context, so one call tears them down cleanly and they rebuild from the new settings. This page runs on exactly this setup.

Per-section setup 
const contexts = new Map();

function initSection(section) {
  // Scoped to this section; only runs if motion is allowed
  const mm = gsap.matchMedia(section);
  mm.add("(prefers-reduced-motion: no-preference)", () => {
    section.querySelectorAll("[data-split-intro]").forEach(splitIntro);
    section.querySelectorAll("[data-word-fill]").forEach(wordFill);
    section.querySelectorAll("[data-sticky-stage]").forEach(stickyStage);
    section.querySelectorAll("[data-marquee]").forEach(marquee);
  });
  contexts.set(section, mm);
}

// SplitText measures text, so wait for fonts first
document.fonts.ready.then(() =>
  document.querySelectorAll("[data-motion-section]")
    .forEach(initSection));
Theme editor events 
// Merchant changed a setting: section re-rendered
document.addEventListener("shopify:section:load", e => {
  initSection(e.target);
  ScrollTrigger.refresh();  // page height changed
});

// Section removed or about to re-render
document.addEventListener("shopify:section:unload", e => {
  contexts.get(e.target)?.revert();  // kills tweens,
  contexts.delete(e.target);         // triggers and splits
});

// Optional: when a merchant selects a word block,
// scroll the stage so that word is showing
document.addEventListener("shopify:block:select", e => {
  /* e.target is the selected block's element */
});
These dropdowns 
function initDropdown(btn) {
  const body = document.getElementById(
    btn.getAttribute("aria-controls"));
  gsap.set(body, { height: 0, autoAlpha: 0 });

  btn.addEventListener("click", () => {
    const open = btn.getAttribute("aria-expanded") === "true";
    btn.setAttribute("aria-expanded", String(!open));
    gsap.to(body, {
      height: open ? 0 : "auto",  // GSAP can tween to auto
      autoAlpha: open ? 0 : 1,     // hides from tab + readers
      duration: motionOK() ? .5 : 0,
      onComplete: () => ScrollTrigger.refresh()
    });
  });
}
Why refresh()?

ScrollTrigger measures where every trigger starts and ends when the page loads. Opening a panel pushes everything below it down, so those measurements go stale and animations would fire in the wrong place.

Calling ScrollTrigger.refresh() after the panel finishes opening re-measures every trigger. The same thing applies to anything that changes page height after load: lazy images, accordions, filters, or the theme editor re-rendering a section.

The toggle is a real <button> with aria-expanded and aria-controls, and it still works with reduced motion on. It just opens instantly.
